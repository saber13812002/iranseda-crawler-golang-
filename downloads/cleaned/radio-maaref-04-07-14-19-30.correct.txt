بر ما بگستر و با رساندن میوه‌ها بر متمدنانت منت گذار و با شکوفایی قنچه‌ها زمین خود را زنده دار. فرشتگان عالی‌مقام و سامن‌بخش را حاضر نما و بفرما تا از سوی تو بارانی آورند سودمند و دامن‌دار، با دانه‌های درشت و پربار، با رگبارهای تند و بی‌قرار، آن‌سان که آن‌چرا مرده است زندگی دوباره دهی و آن‌چرا از دست رفته است بازگردانی و هرچرا روئیدنیست برویانی. و مردمان را فراوان رو زیدهی با عبری انبوه و شادیاور، بیزیان و بیکران و بسی خروشان، که نه سیل پدی دارد و نه برقش تنها بفریبد.
پروردگارا، بر ما بارانی ببار که قهقری‌زدا و سبزی‌زا، فره‌بخش و بیکران و فراوان باشد، تا با آن قامت گیاهان روینده را استوار سازی و پجمرده‌ها را سرفراز.
خدایا، بر ما چنان باران که از تپه‌ها آب جاری کنی، چاه‌ها را سرشار سازی، جوی‌ها را روان و درختان را برویانی و در همه شهرها ارزانی پدیداری، چارپایان و مردمان را شادمان داری و روزی‌های پاکیزت را بر ما کامل گردانی، که آشتگاه‌های من را برویانی و چهارپایان را پرشیر سازی و بر نیروی من بیفزایی.
خدایا، چنان مکن که سایه آن ابرها بر ما باد سوزان، شبد و سرمایش گزنده و شون گردد و چنان مباد که بارانش را برای ما چون کیفر، ساند باران سازی و آبش را در کام ما تلخ و ناگوار گردانی.

سلام و ادب و احترام خدمت شما شنوندگان محترم برنامه‌های رادیو معارف. با مجموعه‌ای گنج سعادت در خدمت شما هستیم. و طبیعتاً آن چیزی که در حوضه اخلاق اسلامی در قرآن کریم و روایات آمده و در واقع زندگی ما رنگ و بوی اخلاقی بخواد پیدا کنه، با نگاه قرآنی و روایی، تراش و زحماتی که جناب استاد حجت‌الاسلام محمد حادی فلاح می‌کشند و بیان سیر و سلوک زندگی اخلاق الهی و توحیدی رو برای ما با توجه به این تراش عمیق و دقیق بیان می‌کنند. ما در برنامه امروز هم در خدمت استاد محترم هستیم که در همین آغاز برنامه خدمتشان عرض سلام دارم.
بسم الله الرحمن الرحیم. من هم خدمت شما و همه عزیزان شنونده سلام عرض می‌کنم. در خدمت درست، خیلی خوش آمدی جناب استاد. با ترجمه‌ای که جناب آقای اطابکی همکار منشیف ندارند، گاهی برای من توفیق است که خدمت شما برسم و شنوندگانم محترم استفاده کنم. از این جهت برای من این وقت به فرصت غنیمت. بحث در باب رذیلت‌های اخلاقی ناظر به زبان بود، دستیدیم به فحش و فحاشی و آن چیزی که در برنامه‌های گذشته به حضور جناب استاد فلاحی اطابکی مطرح شد، کلیاتی از بحث فحش و فحاشی بود و حقیقت فحاشی از منظر قرآن کریم و روایات مطرح شد. علل و عوامل این بحث اشاره شد و آثار و عواقبی که این رذیلت اخلاقی به دنبال می‌آورد، مواضعی بود که در مجالس گذشته جناب استاد فلاح مطرح کردند. من قبل از این که جناب استاد بحث جدید خودشان را در حوضه ادامه مباحث فحش و فحاشی ادامه بدهند، به مطرح کنند این نکته را با توجه به این که برخی از شنوندگان محترم تماسی با روابط عمومی رادیو معارف داشتند و توی تماس خودشان این نکته را اشاره کردند، من تره بحث می‌کنم تا ان‌شاءالله با توضیحاتی که جناب استاد فلاح را مطرح می‌کنند، رفع سوء برداشت و سوء تفاهمی که در باب این مسئله مهم در حوضه رذیلت‌های اخلاقی برای این دست از مخاطبان جوان شکل گرفته، ان‌شاءالله صورت بگیره.
جناب استاد، پاره‌ای شنوندگان این بحث را اطلاع کردند که ظاهراً در برخی از روایات ما مجوز برای فحاشی داده شد. البته برنده که ندیدم ولی خب بر اساس این چیزی که شنوندگان اشاره کردند، حالا آیا واقعاً داریم چنین روایاتی یا نه؟ نکته دیگری داره که نیاز به توضیحی داره، خلط بحثی صورت می‌گیره با مسئله دیگر و نکته دیگری. این رو اگر یه توضیح اجمالی بفرمایید، بعد ان‌شاءالله ادامه مباحث رو در باب مباحث ناظره به فحشی که داشتیم، ان‌شاءالله از زبان شما بشنیم.
در صحبت که منم مثل شما چنین چیزی ندیدم، ولی از صحبت تعالیم توحیدی حضرات معصوم سلام‌الله‌علیهم‌اجمعین، همه اجمعین به دوره هزارات بدزبانی رو نپسندیدند و اجازه ندادند. هیچ‌جا، حتی در باره حقوق خودشان. حالا بعضی‌ها می‌گن که چنین چیزی هست، بنده با چنین گذاره‌ای مواجه نشده و ای کاش این تعالیمی که بیشتر از اون مواردی که مدعی هستن وجود داره، این تعالیم تربیتی و اخلاقی بیشتر در بارش صحبت می‌کردیم، بیشتر در بارش فکر می‌کردیم، عمل می‌کردیم، پیاده‌اش می‌کردیم. ما تربیت اهل بیتی رو کنار بگذاریم و دست کم بگیریم، ما بیاییم تو ساحت زبان، اونم که توصیه نفرمودند و حتی اجازه ندادند، اون بخشش رو بگیریم که بخواییم مثلا این مسیر رو در دفاع از اهل بیت بریم.
جناب آقای رجعی عزیز، ما تالا هرچی که گفتیم با جستجو در آیات و روایات، ما هیچ نظر شخصی خودمون رو اینجا اعلام نکردیم و حقیقتاً من در آیات و روایات جواز فحش و فحاشی نیافتم. در روایات از انات محسوم سلام‌الله‌علیه مجمعی نیافتم. در سیره اهل بیت همین‌طور. و اونایی که کمل در دائره تربیتی اهل بیت علیه‌السلام مصطلح تربیت شدن در سلوکمون، بزرگان و سیره بزرگان هم ندیدم و نیافتم.
این قسمت بحث چون اشاره شد، من گریز بزنم. این رو از حضرت آیت‌الله جوادی آملی شنیدم توی بحث تفسیریشون که خدا حتی نسبت به حیوانات و محیط هم چنین احانت و فحوشی رو اجازه ایراده تشریف نداشتید. ما گفتیم تمامی روایات رو خوندیم و جناب آقای اطابکی عزیز بودن و من اونجا گفتم: «یوش، هواش روایات رو بخونیم، شگف زده میشیم که اینا رو عزیزان واهز و خطیب سر منبرها به مردم نگفتن.» ایند بفرمایید و خیلی دیر کردیم در انتقال تعالیم ناب توحیدی. ولی حتی به آب و هوا بد نگید، به ایوانات بد نگید، به کو دشت سهرا بد نگید که به شما برمی‌گرده. خیلی مهمه، به شما برمی‌گرده. و ما در آثار و هواقب خیلی خواستیم برجسته کنیم، چون آدم‌ها محملاً حب نفس دارن و خیلی نگران هنگه به خودشون آسیب نرسه. مهمترین عقوبت در فحاشی این بود که اون فرش به خود اون آدم برمی‌گشت. به خاطر خودش هم شده، ای کاش فرش رو ترک می‌کرد و از اون عواقبش و اون آثارش در امان می‌موند.
بله، توجه باید اومده همون جور که شما مشارع کردید. به نظر جناب آیت‌الله جوادی آملی حفظه‌الله، امیدواریم که جامعه سالم از هر گونه آسیب‌های زبانی و رفتاری و تو ساحت عقیده و عقل و عقیده و هر سواد ایمان و قلب و اخلاق و فضاهل و همچنین رفتار و سبک و سلوک زندگی مطابقه با خواسته و میل و طلب حضرات معصوم سلام‌الله‌علیه‌مجمعین داشته باشید ان‌شاءالله.
خب، این تحریزی بود به بحثی که الان احیانا شاید سوی برداشتیه، ان‌شاءالله سورت یرفته باشه در باب بحث امروزمون. در ادامه بحثایی که در باب مسئله فوخش داشتیم، ظاهران وعده به بحث راهکالهای علاج انسانهایی که هیانند بچارش شده باشن. خب، من این نکتر هم اضافه کنم: کلن ما اگر میریم سراغ آیات و روایات و بحث خودسازی رو میکنیم، داریم ناخل این تعلیم و آموزها هستیم. هیچ ادعایی در خصوص این تعالیم در باره خودمون نداریم که در ما مثلا پیاده شده باشیم. یک پیشنهادیه از متن شریعت برای تربیت. بعضی‌ها زاقش اونجور در نمیاد یا میل ندارن یا دوست ندارن. هر کس هر جور دلش میخواد تربیت میشه، خودش هم دامن میزنه به تربیت شدن خودش. ما داریم پیشنهاد میدیم توی متن شریعت چنین دهوتی اتفاق افتاده.
خب، راهکارها رو یه قدری مفصل عرض می‌کنم، چون قبلن عرض کردم همه باید به پاک شدن جامعه از فخش و فحاشی کمک کنیم. لذا اگه باز به بعضی از آیات و روایات اشاره می‌کنیم، تکرار نیست، قصد تکرار نداریم. به این خاطر اشاره می‌کنیم که برای راهکارها به کارمون میاد. این گزاره‌ها و این اصناد. خب، این راهکارها تکرار می‌کنم از قبل ذهنم بود این جمله رو بگم: چقدر به جا. این راهکارها به درد کسی میخوره که میخواد فوش و فهاشی رو کنار بذاره. خب، تعداد شده خدمتون عرض میکنم. خب، البته اول به تناسب علال و حواملی که ذهن کردیم و شما تشریف نداشتید، راهکارها رو ناظر به همون یاد میکنم. بعد میام به تناسب از بیان و سیره هزارات و محسون با عنوان نمون راهکار یه چند تایی رو هم ذکر می‌کنم.
اما راهکار اول: جهل رو به آگاهی و یخزه و بیداری تبدیل کنیم. مثلا در باره آثار و عواقب فوش و فحاشی طرف بیاد فکر کنه. فکر کنه این که فرمودن فوش به خودش بر می‌گرده، در واقع خواستن که به خودش هم فوش نده. و فکر کنه این که فرمودن فوش، مثلا فوش ناموسی نده، مال اینه که فوش ناموسی مثلا به مادر و خاهر دیگری به مادر و خاهر خودش برمیگرده. در واقع خواستن هم حقیقت فحاشی رو در گزاره‌های روایی از راتم حسون سلام‌الله‌علیه ام مشخص کنن و هم خواستن که بعضا اگه کسی مادر و خاهرش مثلا براش مهمه، به خاطر مادر و خوهرش هم شده به ناموس دیگران هم فرش ندی. ولی میشه همون که تو روایت اومده دیگه فرمودن: «لا یوبالی ما قال ولا ما غی لفی»، یعنی بر جسته فرمودن که آدم‌هایی که اینجوری اصلا لا یوبالی هستن، براشون مهم نیست چی میگن، چه فرشی میدن، یا چی درباره اونا میگن، یا به فرمایی این فرشی که دادن به خودشون هم برمیگرده، براشون نزدن مهم نیست. خب، تکرار میکنم این راهکارها به درد کسی میخوره که نمیخواد لایوبالی بشه یا باشه و این راهکارها به درد کسی میخوره که میخواد از لایوبالیگری توی حوضه زبان و رضایل زبانی در بیاد.
مورد اول چی بود؟ جهر رو تبدیل به آگاهی و یغذه کردن.
مورد دوم: درکم رو نسبت به عرضش‌های دینی و مذهبی و ملی و انقلابی تصویر کنن و ارتقا بدن. چون بعضی از جا شما تشکیل نداشتید، تلقیات غلطی داریم و بترفصدن‌های ما مالون برداشت نادرست و تلقیات نادرست و غلطی یکی از ارزش‌های دینی مذهبی ملی و انقلابی داریم. مورد سبب قبل از افت زبانی روی افت ذهنی و قلبی منکار
