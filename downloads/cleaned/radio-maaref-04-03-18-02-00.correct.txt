یک عامل مهم دیگر الان وجود دارد که معمولاً از آن قفلت می‌شود، آن هم عامل تاریخ‌سازی بوده که در دو سده اخیر اتفاق افتاده. ببینید مستشرقین از فعالیت‌های زیادی برای نوشتن تاریخ شرق انجام داده‌اند و الان هم شما اگر با تاریخ سر و کار داشته باشید، یکی از مهم‌ترین منابعی که تحصیل گذارده‌اند، نوشته‌های مستشرقین است. مستشرقین در این تاریخ‌نویسی صادق نبودند؛ حتی به تاریخ‌سازی پرداختند و تاریخ را آنگونه که اتفاق افتاده بازگشتند. مثلاً در مورد احامنشیان یک اموری را گفتند که واقعیت ندارد یا اموری را مخفی کردند که آن هم باید گفته می‌شد. این باعث شده که یک فضای دروغین در مورد باستان ایجاد بشود. مثلاً میان می‌گویند که احامنشیان زرتشتی بودند، در حالی که شما برید همه منابع را بگردید، می‌بینید که اثری از زرتشت و زرتشتی‌گری در دوره آن‌ها نیست. حتی بعد از احامنشیان در دوره اشکانیان هم اثری از زرتشتی‌گری نیست. هخامنشیان احامنشیان همشون مقبره دارند. ما قبرستان‌ها و گورها و تابوت‌های زیادی از دوره اشکانیان که بعد از احامنشیان هستند داریم. این‌ها نشان می‌دهد که مرده‌هایشان را دفن می‌کردند در حالی که زرتشتی‌ها این کار را نمی‌کنند. یا در کتیبه‌هایی که خودشون آدانی کنید، اثری از زرتشت نیست.

شوربختانه به این اکتفا نکردند و به وسیله تحصیل‌کردگان غربی داخلی و غرب‌زده‌های وطنی به نگارش‌ها و گزارش‌های مستشرقان عرضش و اعتبار بخشیدند و به قول آقای ادوارد سعید، اسناد شرق‌شناسی وقتی معتبر شناخته می‌شد که از زیر آتش پالایش غربی‌ها عبور کرده باشد. استعمارگران غربی نه‌فقط به وسیله ارتش، بلکه با کمک ادبیات، مردم‌شناسی، باستان‌گرایی و توجیحات روانی شرق را استعمار و استثمار کردند. باستان‌گرایان امروزی نیز با استناد به منابع مستشرقان و با هدف الگوگیری از دوران باستان و چهره‌ای ملی در صددند تا نظام و ساختاری بر اساس تمدن صرفاً ملی بنا کنند، در حالی که باستان‌گرایی بیشتر در قالب توجیه نمادها، استورها و نیز آثار تاریخی بر جای مانده از دوران باستان همچون تخت جمشید و کروش صورت می‌گیرد که فاقد هرگونه اندیشه‌ی الگوسازی است.
دکتر رزوی در مورد خود هخامنشیان، الان مهم‌ترین نمادی که از هخامنشیان معرفی می‌کنند، تخت جمشید است. بعد یک داستان سر آیه‌هایی می‌کنند که در اینجا زرشن‌های فلان برگزار می‌شود، فلان می‌شود. در حالی که شما وقتی رو زمین کنید می‌بینید که اصلاً این تخت جمشید که ساخته شده، هیچ‌وقت به بهره‌برداری نرسیده. در آن همون مرحله ساخت نیمه‌کاره رها شده. اصلاً هیچ‌وقت کنتگاه به منایه واقعی نبوده. بین‌خب باستان‌شناسان غربی وقتی که اینجا را می‌خواستند احیا کنند، بعضی چیزها را اضافه کردند، بعضی چیزها را کم کردند، حتی بعضی کتیبه‌ها افصودند. یعنی این سوءاستفاده از باستان برای زلی دادن خوب باستان باید شده که تصورات غلطی بین مردم و کسانی که گیرشان به باستان دارند پیدا کنند. اسم‌ها را عوض کردند، مکان‌ها را عوض کردند، چه‌هایی را به عنوان مثلاً قبر فلانی کادشاه معرض می‌کردند که اصلاً اینجا نبوده.
پس ببینید، یکی از عوامل باستان‌گرای امروز، آن جل و تاریخ‌سازی بوده که مستشرقین یکی از عواملش هست. دومین عامل هم خب این است که نقطه قوت انقلاب ما همین دین اسلام هست. طبعی است که مخالفان به آن نقطه قوت حمله می‌کنند. یکی از عواملی که می‌توانند نقطه قوت ما را مورد هدف قرار بدهد، بحث‌های نژادی و باتره بحث‌های نژادی و این است که این دین، دین اسلام، مال عرب‌هاست و دین آریایی‌ها دین زرتوشتیه. بخواهند که بین باستان و حال تفرقه ایجاد کنند، در حالی که ما می‌دانیم بهترین و شکوه‌مندترین دوره تاریخی امروز بعد از ورود اسلام بوده. از نظر تمدنی ما هیچ‌گاه به اندازه‌ای که در دوره اسلام دارای تمدن بودیم، نیستیم. آن‌های فراوانی هم می‌شود شمرد. نه این که بخواهم مثلاً دوره قبل از اسلام را نفی کنم، ولی خب در مقایسه، شما نمی‌توانید مثلاً دوره شکوه‌
