و خوب، خسارات جدی مالی، جانی و به مراکز مهم علمی و نظامی، این‌ها همه خسارات جدی به آن‌ها بود. و بعد، آن‌ها تصور می‌کنند با پنج یا شش روز حرکت نظامی می‌توانند همه‌چیز را عوض کنند. خب، به هیچ‌وجه نتوانستند. نه فقط نتوانستند، بلکه شرایط در آنجا برای مردمشان خیلی سخت‌تر شد؛ وضعیت معیشتی، محاصره، وضعیت ناامنی که آنجا اتفاق افتاده و خسارات سنگینی که آره، دوزخ‌هایش آشکار است و فیلم‌هایش دارد بیرون می‌آید. واقعاً گاهی آدم احساس می‌کند آنجا شبیه غزه شده؛ یعنی آن تخریب‌هایی که آن‌ها در غزه کردند، گاهی شبیه آن‌ها در بعضی مناطق اتفاق افتاده. بعد، حجم سرزمین‌های اشغالی خب خیلی کم است، لذا این تخریب‌ها خیلی خودش را نشان می‌دهد و فرق می‌کند با ایران. لذا این‌ها خسارت‌هایی است که به آن‌ها وارد شد. به احزابشان دست ندادند، بدان نکردند و اصلاً بهترین علامت این که آن‌ها شکست خوردند، این است که دیگر همه می‌دانند آن‌ها اومدند. اولاً آمریکا وارد شد. چرا آمریکا وارد شد؟ چون استرالیا احساس کرد نمی‌تواند تنهایی ادامه بدهد. خود ورود آمریکا مثل بچه‌ای است که در دعوا زورش نمی‌رسد، می‌رود بزرگ‌ترش را هم می‌آورد. خود ورود آمریکا یا حمل مستقیم آمریکا به بخش‌های هسته‌ای ما، علامت این است که آن‌ها دچار ضعف شدند. و بعد، سریع درخواست آتش‌بس. خب، درخواست آتش‌بس از موضع ضعف، دیگر. لذا این‌ها همه نشان می‌دهد که آن‌ها دچار خسارات سنگین و مشکلاتی شده‌اند که می‌خواهند جبران کنند.
و از آن طرف، آورده‌های ما و دستاوردهای ما چه بود؟ ما از سرزمین هم دفاع کردیم، نسبت به خسارات، من‌تلافی کردیم. در ابعاد بین‌المللی، خصوصاً در بین ملت‌های دنیا، چقدر ایران عزیز شد؟ چقدر مقام معظم رهبری محبوبیت بیشتری پیدا کرد؟ دارید شما بینید، در فضای محبوبیت کلیپ‌های متنوع دارد پخش می‌شود. چقدر وحدت ایجاد شد؟ علمای اهل سنت ایران و خارج از ایران، الازهر مصر، در دفاع از جمهوری اسلامی، در دفاع از شخص رهبری، حکم محارب دادن؛ بعضی‌شان به‌درد ترامپ و کسانی که با آمریکا و رژیم صهیونیستی همکاری می‌کنند. بعد، اپوزیسیون‌های داخلی و خارجی اومدند حمایت کردند. دیدگاه‌هایی که علایق فکری، گرایش فکری، سیاسی و عقیدتی‌شان باز هم با جمهوری اسلامی متفاوت بود، اومدند حمایت کردند. اصلاً وحدت عجیبی که اتفاق افتاد بین مردم و احیای مجدد آن روحیه ایثار و فداکاری و شهادت‌طلبی که در دوران دفاع مقدس بود، دوباره این احیا شد. و ما دیدیم در بین مردم همکاری‌ها و حمایت‌هایی که مردم نسبت به هم‌دیگر کردند و از آن طرف دولت قدرتشان را نشان داد که کاملاً توانست، مثلاً فرض کنی سخت و تأمین بکند، مواد اقتصادی و تأمین بکند، شما احساس نکنید چیزی کم است. و نیروهای نظامی هم قاطعانه پاسخ دادند. خب، بله، طبیعتاً جنگ خسارت‌هایی دارد و دستاوردهای این به اصطلاح دوازده روز، دستاوردهای مادی و معنوی فراوانی برای ما داشت. و عملاً در حقیقت جمهوری اسلامی یک ارتقاء جدی در دنیا پیدا کرد؛ هم از لحاظ محبوبیت، هم از لحاظ احساس قدرت. و آن‌ها هم واقعاً دیدند نه، به قول معروف، این توپ‌میری‌ها، آن توپ‌میری‌ها نیست؛ یعنی این ایران آنی نیست که آن‌ها توپ‌میری کرده‌اند. نه مردم آنی بودند که آن‌ها فکر می‌کردند به این راحتی می‌توانند از اختلافاتشان استفاده کنند. نه نیروهای نظامی توانش آن‌جوری بود که این‌ها فکر می‌کردند؛ چون گفته بودند دیگر اندیشیده‌اند و آن‌که گفته بودند ایران از لحاظ نظامی ضعیف شده، بازوهای منطقه شاید دست داده و از آن می‌توانیم سریع سرکوب و منکوبش کنیم. و خب، هیچ‌کدام این‌ها اتفاق نیفتاد و بلکه عکس شد و ما آورده‌های بیشتری پیدا کردیم و وحدت ملی‌مان بیشتر شد. به قول آن خانم اپوزیسیون در خارج از کشور گفت: بیست سال براندازی در ایران عقب افتاد. اگر این‌ها امیدی داشتند که از داخل حرکتی بشود، این عملاً از بین رفت. و واقعاً باعث تقویت، انسجام، استمرار و حیات بیشتر جمهوری اسلامی و مردم ایران شد. بله، حالا شهدایی دادیم. و البته این شهدایی هم که ما از دست دادیم، در یک نگاه قرآنی، اهل حسنین، این هم خسارت نیست؛ بلکه آن‌ها به مقامات بلند معنوی رسیدند که خوش‌حالشان بسیار خوب است.
خیلی ممنونم. آقای دکتر عبود طالبی عزیز و عرکمند، همچنان با شما هستیم با برنامه پرس‌و‌پاسخ انقلاب از رادیوی معارف. شنوندگان گرامی، شما با برنامه پرس‌و‌پاسخ انقلاب همراه هستید. گفتگو را ادامه می‌دهم و تره سوال‌های شما از حجت‌الاسلام‌والمسلمین دکتر عبود طالبی، کارشناس و تحلیلگر مسائل سیاسی. پیامک فرستادن به ۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵۳۵
