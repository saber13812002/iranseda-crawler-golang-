بودن که به اصطلاح سرِ توپُک بودن و شکاف طبقاتی، مثلاً در آنجا آن‌طور معنا شده بود. و این‌ها آن‌هایی بودند که مرتبط با حکومت بودند، شاخ بودند و این‌ها در حقیقت از این وضعیت مناسب بهره‌مند بودند؛ خانه‌های ویلایی داشتند، جاهای ویلایی داشتند. و زندگی امثال شخص بنده، زندگی در همان حد متوسط است؛ همان حد متوسط وجود دارد. و این‌ها یعنی در حقیقت نه جایی ویلایی داریم، هیچ چیزی در حقیقت مازاد. در آن اقلیت، حتی مثلاً الان من ماشینی به نام خودم ندارم؛ یعنی همین ماشین به عنوان مسئول در اختیار ماست و این‌ها در اختیار هستند و آن‌ها شب‌ها ماشین را می‌برند و می‌آورند، تو همین مجموعه. و این‌ها داماد ما چرا ماشین دارد، ولی من خودم شخصاً ماشین ندارم؟ این‌ها سعی کردیم که در حقیقت این زندگی را در این حد متوسط نگه داریم؛ از پخت‌وپز و خوردن غذای ساده و ساده‌زیستی و ساده خوردن و این چیزها را سعی کردیم. شاید هم به نسبت بعضی‌ها که بالاخره هنوز این امکانات را نداشته باشند، وضعیت ما خیلی بهتر از آن‌ها باشد؛ خیلی بهتر از آن‌ها باشد.
آیت‌الله رمزانی گیلانی از دوران کودکی علاقه بسیاری به کتاب و متعلقات داشت. او به جهت متعلقات فراوان، در پاسخ‌گویی به پرسش‌های دیگران موفقیت‌های چشمگیری داشت. من او وقت‌ها خیلی متعلقه می‌کردم، خصوصاً بعد از انقلاب؛ چون انقلاب که من ۱۴ سال داشتم، تقریباً همون ۱۴ سال، ۱۴ سال و نیم این‌ها بودم. ولی بعد از آن، وقتی که در فضای مسجد و این‌ها قرار گرفتیم، خیلی خیلی به متعلقه من شدیم. خیلی داستان‌های پیامبران را می‌خواندیم، قرآن‌ها را می‌خواندیم. گاهی کتاب‌های مختلف درات‌های جوانان نوشته شده بود، آن‌ها را متعلقه می‌کردیم. علاقه شدید به کتاب داشتیم و خودبه‌خود خیلی از سؤالاتی را که دیگران می‌کردند، ما پاسخ می‌دادیم و اهل متعلقه بودیم. بعد کم‌کم این جریان چریک‌های فدایی، منافقین خلق و این‌ها که بجور بجور آمده بود، و این‌ها ما کم‌کم با این‌ها هم بحث می‌کردیم. و لازمه بحث کردن این بوده که ما باید خودمان را محیام می‌کردیم؛ برای بحث مطالعه می‌کردیم. و خودبه‌خود وقتی که مثلاً می‌گفتند بحث می‌گفتند فلانی بیاید کمی بحث بکند با این‌ها، و وقتی که کتاب می‌خواندیم، قصه‌ها را می‌گفتیم یا حدیث می‌گفتیم، در جاهای دیگر استفاده می‌کردیم. خودبه‌خود اصلاً انسان گویا این‌که در این مسیر دارد جهد داده، می‌شود که باید گوینده بشود، باید حرف بزند؛ چون آن‌ها دارند گوش می‌دهند. یعنی این مسیر بوده. و حتی من یادم هست، من در دوره دبیرستان خیلی جالب هستم، می‌شود سال مثلاً ارسالت که پنجاه و شش و سن، یک مدیری داشتیم. بنابراین آقای نیازمت‌ها زگی فوت کرد. آدم متدینی بود، خیلی متدینی. اصلاً خانوادشون خانواده‌ی اصلاً بچه‌های انقلاب هستن. برادر و برادرزاده‌هایشون... من در آنجا امام جماعت شده بودم؛ یعنی اوورکوت داشتم. آن وقت‌ها یک اوورکوتی هم می‌پوشیدند؛ از این اوورکوت ایرانی‌ها را می‌پوشیدیم. این‌ها خیلی اکس هم باشده دادم. حتی من یادم هست که یک زبان همین اوورکوت را روی من پوشیده بودم، تو زندان رفته بودم. از طرف مهم آقای شرفی، آنجا سخنرانی می‌کردم. اکسشم دارم. این‌ها اکس، این به اصطلاح سخنرانی برای زندانیان را که در آنجا رفته بودیم، بخوادر همین مطالعات. و حتی من همون زمان تو محله خودمون منبر می‌رفتم. بعد دیگه کم‌کم عبا گذاشته بودیم. این‌ها تو مسجدم، یک آقای امام جماعت ما بود؛ الان در غیر حیات هست. تو همون رازی، تو قاسمیه. گاهی ایشون که نمی‌آمد، می‌گفت: «فلانی نماز بخواند، فلانی نماز بخواند.» احساس می‌کردند یک مسئولیتی دارم، متوجه من می‌شدم. یعنی ایشون وقتی که می‌گفت: «من نماز بخوانم»، معلوم بود که منو قبول دارد و معلوم بود که یک صلاحیت‌هایی را در من می‌بیند. از این روحانی که من وارد مجلس می‌شوم، ایشون از منبر کاملاً می‌ایستد؛ یعنی وقت وارد مجلس که شدم، دیدم کاملاً در حال سخنرانی بود. استاد احترام کرد. از اون وقت به ما این علاقه را داشت. بعد من این عبایی گرفته بودم از یک روحانی به نام آقای معده‌پور. اون روحانی اون عبا را که به من داد، من صبح‌های ماه‌ها ماه رمضان، چون دیگه ما رمضان مدارس تعطیل بوده، در این‌ها تو محل خودمون که بودیم، نماز جماعت را ما می‌رفتیم. این سال مثلاً ۶۲، ۶۳، ۶۱، نماز جماعت اونجا را ما می‌خواندیم. پیرمرد می‌آمدند، جوان‌ها می‌آمدند، آن‌ها ما رو قبول داشتند.
این‌ها یکی از مباحث مهم در حوزه‌ی آموزش و پرورش، نوع رفتار با دانش‌آموزان است. در دوران قبل از پیروزی انقلاب اسلامی، رفتار نظام آموزشی با دانش‌آموزان بیشتر جنبه تنبیهی داشت؛ اما پس از پیروزی انقلاب اسلامی این نگاه اصلاح شد. آیت‌الله رزا رمزانی از دو نظام تربیتی در قبل و بعد از انقلاب اسلامی می‌گوید: من خاطرم هست، حالا دوران کودکی را که می‌گویید، این دیگه بیشتر برمی‌گردد به اون ابتدایی و راهنمایی. و دیدم خیلی‌ها کتک می‌خوردند. تو بین آدم‌های شیطان هم تو کلاس بودیم؛ افرادی که خیلی شیطان بودن، عذاب می‌کردند. این‌ها لذا مدیران و اونایی که نظام مدرسه بودن و گاهی معلم بودن در کلاس، یکی از راه‌های تنبیه را همین تنبیه بدنی می‌دانستند. تنبیه می‌کردند، گاهی این‌ها تشر می‌دادند، دعوا می‌کردند، بیرون می‌کردند، اخراج می‌کردند. ولذا این‌ها ساکت می‌شدند، می‌ترسیدند؛ چون اگر این‌ها را به این تاریخ ساکت نمی‌کردند، اصلاً مختل می‌کردند کلاس را، مدرسه را. و به همین جهت از این‌ها می‌ترسیدند و حساب باشد، این‌ها آدم‌های قدرتمندی بودن؛ چون قدرتمند بودن، در مقابل این‌ها هم نمی‌تونستن به اینستن، به مقابله بکنن. شاید به اختصار اون زمان، حالا چون خودم به اصطلاح تو حالا حوزه‌های تربیتی یک سری متعلقاتی دارم، کارهایی دارم، شاید حالا شاید یکی از تخصص‌هایی که هست تو همین حوزه‌های تربیتی باشد. و این‌ها تنبیه را همیشه یک عمر مغلوب میدانم، نه خالب. تو عرصه تربیتی، اون‌ی که باید تو عرصه تربیتی بیشتر مورد نظر مربی قرار بگیره، باید تشفیق باشه و راه‌نمایی باشه. اون وقت‌ها خیلی ما راه‌نمایی نمی‌دیدیم؛ یعنی این‌طور بیا دلسوزانه بنشینه حرف بزنه، به عنوان یک مشاور امین. اون زمان اصلاً مشاور نبوده، اصلاً مشاور در مدارس نبوده. راحت‌مان خودتونم یادتون هست، اصلاً مشاور در مدارس نبوده. لاغل تو زمان ما نبود، قطعا دوری قبل از ما هم تو این مدرسه‌های عمومی اینجور چیزا اصلاً نبوده که بیان مثلا مشورت بدن یا مثلا گاهی تذکرات لازم رو بدن. و امسال زنگوینام، اونا به خاطر این که گاهی از نقی تو مدارس میدیدن که دو نفر هستن که اختلال وارد کردن در فضای مدرسه و دران مدرسه رو به اصطلاح به آشوب میکشن، گاهی مجبور میشدن که اینها رو تنبیه بدنی داشته باشن. ولی تنبیه بدنی در حقیقت در عمر تربیتی امروز، با توجه به شرایطی رو که داریم، پاسخگونیست. ما باید تمام تلاش ما با تشفیق و موئزه و پند و راه‌نمایی و مشورت و مشورت دادن باشد. و این‌ها ما باید متناسب با شرائط امروز هم در حوزه‌های آموزشی، متناسب با فضا، کتاب آموزشی رشته باشیم که بتونه پاسخگو باشد. و این‌ها هم تو حوزه‌های تربیتی امروز. خب معبولت تحصیب داریم، امروز بالاخره مربی هست، امروز مشورت هست، امروز خیلی کارهایی انجام دادن. حالا به این استاندارد‌ها ما نرسیدیم، خیلی فاصله داریم با استاندارد‌هایی رو که تو حوزه‌های تحزیبی و تربیتی لازم هست که باید پیگیره بشه. ولی در این حال، پیشرفت‌های بسیار بسیار خوبی ما الحمدلله امروز داریم. و این‌ها به استانداردش خیلی فاصله هستیم، به اون مطلوب رسیدن و این‌ها. و در از را خودم چون که از کردم شیوه‌های تربیتیی روی که مثلا در امریکا هست، در اروپا هست، در ایران هست، در جاهای مختلف هست، این‌ها کم و بیش با این شیوه‌های تربیتی آشنایی دارم. این‌ها و اونجوره که در خود اسلام هم هست، خصوصا تو دروهای هفت ساله، مثلا هفت ساله اول، دوم، سبوم، از اون هفت ساله‌ها استفاده نمیشه که تنبی قومان یک شیوه تربیتی باشد. و این‌ها.
همنشینی با بندگان صالح خدا، لذت وصف‌نپذیری دارد و حسی دلنشین. آیت‌الله رمزانی از حضور خود در جمع پیروان خط امام می‌گوید: ما رفته بودیم، آشنا شده بودیم. خوزن یک پیروان خط امام داشتیم. در مسجده به اصطلاح کاسفرشان که آی جوازداده بود، کلان فوت کردن، خیلی صدای خیلی صدای قشنگی داشت. بعد یک دوره نماینده شده بود، به اصطلاح دای کومل‌های بسیار زیبایی داشته. منم علاقمنده به دای کومل شده. بعدها دیگه خاطر من، چون صدا من بد نبود، با اون دای کومل میخاندم. تو اون حال هوای دعا و گایی هم از حال میرفتم؛ یعنی انقدر دعا رو دوست میداشتم، گوش دادن دعا و احساس بکردیم که تو این مسیره باید دادم قدم بردارید. مجموعی اینا بله، همینطور هم کتاب با امان رفیق هم اون جمعی رو که تو اون مسجد بود، هم اون فضایی بسیار بسیار خوبی که ما در اون پیروان خط امام داشتیم که در عرض مثلا یک سال چیش هفت از این بچه‌ها شهید شده بودن. اینقدر اینها رو دوست میداشتیم، اصلا خود بخود اون فضا و اون معنویات در ما یه تأثیر بسیار بسیار امیغی داشتد و اینا بسیار عمیق داشته.
بیس و سه و چهل دقیقه است و دقایق پایانی روز دوشنبه که هر روز رو با دعای اهد امام زمان آغاز میکنیم و با زیارت آل یاسین به پایان میبریم و لحظات دیگه با نوای جنابای مهده هیدرزاده زیارت آل یاسین رو زمزمه خواهیم کرد و ارز عدب به صحبت مقدس آقا جانمون اولامون امام زمان علیه السلام.
سلام بر مهربانترین، سلام بر شکوفاترین. اینک با چشمهای پرست تاره و دستهای برنگ تمنا به زیارت آل یاسین می رویم و بر آنان سلام و درود می فرستیم.
