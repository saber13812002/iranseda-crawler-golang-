افغانستان را ترک کردند و چقدر در همین خروجشان از افغانستان، آسیبی به این کشور مظلوم زدند. اولین هدفی که امام راحل (ره) پیش می‌گرفتند، مسئله‌ای بود که ساختارهای سلطه‌گری خارجی از ایران بازگردند. هدف دوم امام این بود که یک دولت اسلامی، مستقل و غیرمتعهد در آن زمان تأسیس کنند. شما شاهد این هستید که در آن زمان، با توجه به اینکه دنیا فضایی دو قطبی داشت (بلوک غرب و بلوک شرق)، قاعدتاً چندان قابل فهم نبود که شما دولتی داشته باشید که نه شرقی باشد و نه غربی. این موضوع حتی برای خود سلطه‌گران آمریکایی هم قابل فهم نبود؛ یا باید وابسته به بلوک غرب می‌بودید یا بلوک شرق. یعنی شما در غیر از جهان دو قطبی، قابل تصور نبودید. امام دنبال این بودند که ایران، که عملاً قبل از انقلاب اسلامی در زیر سلطه بلوک غرب قرار داشت، بعد از انقلاب اسلامی به صورت غیرمتعهد درآید. نگفتند: «حالا انقلاب شده و چون انقلاب در آن زمان بیشتر ریشه در تفکرهای شرق‌پایه و نگاه کمونیستی داشت، پس شما از سمت غرب می‌خواهید به سمت شرق بروید.» تلاش امام این بود که الگویی از یک دولت اسلامی، مستقل و غیرمتعهد به تصویر بکشند.
سومین هدفی که امام راحل (ره) در مواجهه‌شان با ایالات متحده داشتند، این بود که بتوانند علاوه بر مردم ایران، ملل مستضعف جهان را علیه نظام امپریالیستی جهانی که به رهبری آمریکا اداره می‌شد، بسیج کنند. یک بسیج عمومی مردمی در ایران و جهان که مرکز تلاش‌شان ایران باشد. در این‌جا می‌توانند یک بسیج جهانی مستضعفین علیه نظام امپریالیستی جهان را به پا دارند. شما مکرراً در تعابیر امام، آیه «وَعَدَ اللَّهُ الَّذِينَ آمَنُوا مِنكُمْ وَعَمِلُوا الصَّالِحَاتِ لَيَسْتَخْلِفَنَّهُم فِي الْأَرْضِ كَمَا اسْتَخْلَفَ الَّذِينَ مِن قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِينَهُمُ الَّذِي ارْتَضَىٰ لَهُمْ وَلَيُبَدِّلَنَّهُم مِّن بَعْدِ خَوْفِهِمْ أَمْنًا ۚ يَعْبُدُونَنِي لَا يُشْرِكُونَ بِي شَيْئًا ۚ وَمَن كَفَرَ بَعْدَ ذَٰلِكَ فَأُولَٰئِكَ هُمُ الظَّالِمُونَ» (آیه استخلاف) را می‌بینید؛ آیه‌ای که امام بارها به آن تمسک می‌کردند و می‌فرمودند خداوند به مستضعفین عالم وعده داده است که اگر اصول الهی را رعایت کنید، پیروز عالم خواهید شد.
نکته‌ای را عرض می‌کنم: تا قبل از انقلاب اسلامی، در دنیا دو اردوگاه وجود داشت: اردوگاه سلطه‌گران و سلطه‌پذیران. انقلاب امام این دو اردوگاه را به سه اردوگاه تبدیل کرد: سلطه‌گر، سلطه‌پذیر و سلطه‌ستیز. این اردوگاه سوم که امام درست کردند، دقیقاً نقطه مقابل مطالبات نظام استکبار جهانی است. آمریکا به عنوان علمدار نظام استکباری در عالم، به دنبال این بود که این اردوگاه سوم را از بین ببرد. هنوز همین خط فرمان در ادبیات استکباری وجود دارد. این‌جاست که عرض می‌کنم امام و آقای خامنه‌ای (ره) یک خط واحد را در نحوه تقابلشان با آمریکا پیش می‌برند.
پس از آنکه با مبانی فکری حضرت امام (ره) آشنا شدیم که برگرفته از قرآن کریم است، و با توجه به سیاست‌های انتخاب‌شده، کاربستی که در کنار آن راهبرد ارزشمند، بتواند خودش را در صحنه منطقه و بین‌المللی نشان دهد، به این نتیجه رسیدیم که مبانی فکری حضرت امام در این موضوع بسیار مهم، بر چه پایه‌ای استوار بود. جناب استاد مشکیباف به سه مبنا و ویژگی اشاره کردند:
اولی، بحث ساختارهای حضور آمریکا در ایران که باید از بین برود. اشاره شد که حدود هشت هزار و چهارصد نفر از نیروهای آمریکایی در ایران حضور داشتند و حقوق آن‌ها نیز از بیت‌المال ایران تأمین می‌شد که این‌ها باید می‌رفتند.
دومی، این که از یک راهبرد بسیار مهمی ما باید در اینجا استفاده کنیم که از تعریف زیست‌بوم بلوک غرب، به یک دولت مستقل، اسلامی و غیرمتعهد تغییر پیدا کنیم.
سومی، این که بر این مردم ایران که حالا انتخاب شده‌اند و در این موضوع کارشان را به پیش می‌برند، ما سایر ملل مستضعف جهان را هم علیه آمریکا بسیج کنیم.
این سه اصل و سه ویژگی را در اندیشه سیاسی حضرت امام که برگرفته از مبانی قرآنی است، به وضوح می‌بینیم: تبدیل دو قطبی به سه قطبی (سلطه‌گر، سلطه‌پذیر و سلطه‌ستیز) که این در حقیقت به مفهوم سیاسی جهان اضافه شد و این اندیشه راهبردی همچنان ادامه دارد.
با شما عزیزان هستیم. همراهان خوب برنامه «راهبرد»، سخنان ارزشمند جناب استاد مشکیباف، کارشناس این برنامه، پاسخ به پرسش‌های شما درباره اندیشه‌های راهبردی حضرت امام و مقام معظم رهبری (مدظله‌العالی).
(بخش تبلیغاتی/سؤال مخاطب)
امروز یک سؤال مخصوص خود شنونده‌ها داریم. آن سؤال را من بگویم خدمت عزیزان، بعد دیگه کم‌کم سراغ تلفن‌ها و پیام‌ها می‌رویم. اما سؤالی که شما قرار است جواب بدهید، سؤال خوراکی ماست. یک قضای اصیل ایرانی، خیلی ساده و سریعی، سوپ‌مانند که می‌دانم الان اگر بگویم همه متوجه می‌شوید؛ با پیاز، گوجه رنده‌شده، آب و تخمه مرغ درست می‌شود. بعضی چیزهای خیلی دیگری هم به آن اضافه می‌کنند. شما قرار است بگویید که چجوری درستش می‌کنید. حالا اسمش چیست که ما بدونیم؟ متوجه شدید؟ البته که ما حالا اسمش را نمی‌گوییم. در شهر شما به چی معروف است؟ داخلش چی می‌ریزید؟ یعنی چقدر متنوع درستش می‌کنید؟ چقدر دوستش دارید؟ قضیه سنتی و قدیمی است. شاید متأسفانه جدیدترها خیلی سراغش نروند، اما قدیمی‌ترها معمولاً در ماه صفر زیاد دیده می‌شود. سامانه پیامکی ۳۵۵۳ در اختیار شماست.
(بخش سلامت)
اما تدابیر سلامتی را بشنوید. به نام خدا، خوردن از اصول مهم حفظ سلامتی و درمان بیماری‌هاست. اگرچه توجه به نوع غذا بسیار مهم است، اما از نظر طبّی، رعایت آداب صحیح غذا خوردن به اندازه نوع غذا بر بدن اثرگذار است. اصول صحیح غذا خوردن:
۱. وعده‌های غذایی را هر روز در زمان مشخصی مصرف کنید تا دستگاه گوارش هضم بهتری داشته باشد.
۲. غذای مناسب را به اندازه مورد نیاز بدن بخورید و خوب بجوید. بهتر است همراه غذا نوشیدنی مصرف نشود. می‌توانید بعد از هر لقمه، قاشق را در بشقاب بگذارید و برای آماده کردن لقمه بعدی عجله نکنید.
۳. از خوردن چند نوع غذا در یک وعده خودداری کنید، به‌خصوص غذاهایی که مدت هضمشان با هم متفاوت است و در یک وعده مصرف نشوند.
۴. بهتر است موقع غذا خوردن حرف نزنید و مشغول تماشای تلویزیون نباشید.
۵. در آرامش و بدون عصبانیت و استرس غذا بخورید و در حال حرکت یا ایستاده غذا نخورید.
نکته مهم این است که بلافاصله بعد از غذا و با شکم پر، این کارها را انجام ندهید: ورزش یا فعالیت سنگین، خوابیدن طولانی، خوردن میوه و نوشیدنی. سالم بمانید.
بله، این هم از آداب غذا خوردن که همکار خوب ما، دکتر بختیاری، این مطالب را گفتند. ایشان هم خیلی سفارش می‌کنند، معمولاً در صحبت‌هایشان که آداب غذا خوردن خیلی در سلامتی، وزن و همه‌چیز تأثیر دارد. اما دوستان عزیزم، منتظرید که شما سؤالات خودتان را مطرح بکنید. اگر می‌خواهید اطلاعاتی به دست بیاورید، در جبهه بیماری و مشکل خاصی، دکتر پاسخگو باشند. سؤالی که مخصوص به خود شماست.
(بازگشت به سؤال خوراکی)
جبهه غذای اصیل ایرانی، ساده، سریع، سوپ‌مانند، با پیاز، گوجه رنده‌شده، آب، تخمه مرغ درست می‌شود و البته کلی مخلفات دیگر که حالا در شهرهای مختلف ممکن است به آن اضافه کنند. خیلی‌ها می‌گویند متفاوت از هم درستش می‌کنند. در شهرهای مختلف، شما به چی این غذا معروف است؟ بعد چجوری درستش می‌کنید؟ دوستش دارید یا ندارید؟ چقدر استفاده می‌کنید؟ همه را می‌توانید به ما بگویید. ۳۵۵۳ در اختیارتان هست. اما تلفن می‌شود.
(پاسخ به مخاطب)
باریکلا، گل محمدی. در غذا گفتن زیاد استفاده می‌کنند. گفتند گل محمدی در غذا زیاد استفاده می‌شود. حالا یک چیزی را کلن بگوییم: هیچ چیز زیاد استفاده نکنیم. این‌که یعنی اینجوری نیست که هر چیزی، به قول سعدی، یک بیتی دارد. حالا اگر هم شعرش را درست بگویم: «هر چه خوردی از خوردنِ پرِ ملال خیزد». دیگه ما ساده‌تر از آب که نداریم. هر چیزی را زیاد روی کنیم در مصرفش، قطعا مشکلاتی برامون ایجاد می‌کند، چون به هر حال هر چیزی یک عوارضی دارد و خب وقتی طولانی‌مدت مصرفشون عوارزش پیشتر خودشون نشون میده.
گل محمدی که از گیاهان معطر، خوش‌بوه و مقوی قوای بدن است، در همه قوا گفته شده؛ هم قوای روحی، هم قوای جسمی. و خب مزاجش به اصطلاح طب سنتی مرکب‌الخواص است؛ یعنی هم نشانه‌های سردی را از خودش بروز می‌دهد، هم نشانه‌های گرمی. خیلی هم می‌گویند گلاب گرم است، خیلی هم می‌گویند گلاب سرد است. به خاطر آن مواجهه‌ای که خودشان با گلاب داشتند، این اتفاق برایشان می‌افتد. ولی واقعیت این است که هر دو خواص را دارد. بعضی وقتا قبضش پیشتر است، بعضی وقتا ملایم‌بودن یا تلگینش پیشتر است که این هم خیلی رایج است. می‌گویند گل محمدی کاشان ملایم است، گل محمدی تبریز قابض است. ولی به هر حال اینجوری است که هم خواصیت قبض دارد، هم خواصیت تلگین دارد. به هر حال گیاه خوبی است.
حالا این که در چه غذایی استفاده می‌شود، خودش بحث دارد، چون بالاخره یک اطره خواصی دارد. شاید به ذائقه خانواده‌شان می‌خورد، ولی زیاده‌روی نکنند. نه، زیاده‌روی‌اش خطا است و واسطه مشکلاتی می‌شود که خب مناسب نیست اصلاً. کلن ما هیچ عدویهی و زیادشو توصیه نمی‌کنیم. درست؟ مثلاً هفته چقدر؟ یکی دو بار استفاده کند. ببینیم، بستگی باز به خود آدم‌ها دارد، بستگی به مقداری که استفاده می‌کند دارد؛ یعنی در هر غذایی چقدر دارد اضافه می‌کند؟ آره، واقعا هفته دو تا سه بار، مثلاً در دو سه وعده غذایی اشکال ندارد، ولی باز هم مقدار زیاد استفاده نکند.
خیلی مچکرم از توضیحات شما و تشکر از شمایی که داری همراهی می‌کنی تلفنی و پیامکی. اجازه بده من یک بار دیگه این غذای اصیل ایرانی سنتی را براتون بخونم که قرار شد شما اسمشو برامو بفرستین. گفتیم خیلی ساده و سریع درست می‌شود، سوپ‌مانند، با پیاز، گوجه رنده‌شده...
